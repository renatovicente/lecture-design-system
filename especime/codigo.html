Descida do gradiente

O que muda nos pesos de um passo para o outro?

Linhas marcadas e numeradas

import numpy as np

def descida(w, x, y, eta=0.1, passos=100):
    """Ajusta w por mínimos quadrados."""
    for _ in range(passos):
        erro = x @ w - y  # resíduo de cada exemplo
        w = w - eta * x.T @ erro / len(y)
    return w
As linhas marcadas são o passo; o resto só prepara o laço.

No limite: 16 linhas

def perceptron(x, y, epocas=10, eta=1.0):
    """Treina um perceptron com rótulos -1 e +1."""
    w = np.zeros(x.shape[1])
    b = 0.0
    for epoca in range(epocas):
        erros = 0
        for xi, yi in zip(x, y):
            if yi * (xi @ w + b) <= 0:  # classificou errado
                w = w + eta * yi * xi
                b = b + eta * yi
                erros += 1
        if erros == 0:
            break
    return w, b
